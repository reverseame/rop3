; mprotect(addr, len, PROT_READ|PROT_WRITE|PROT_EXEC) -> make a page RWX
; (disable NX/DEP on the target region, then jump into it)
lc(x0)           ; x0 <- addr     page-aligned base
lc(x1)           ; x1 <- len      region length
lc(x2)           ; x2 <- 7        PROT_READ|PROT_WRITE|PROT_EXEC
raw([mov, svc], [(x8, 0xe2), (0)], [], [x0, x1, x2])
